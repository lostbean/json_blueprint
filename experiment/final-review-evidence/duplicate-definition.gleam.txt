import gleam/io
import gleam/string
import json/blueprint/codec
import json/blueprint/value
pub fn main() {
  let duplicate = {
    use a <- codec.field("x", codec.int(), get: fn(v) { v.0 })
    use b <- codec.field("x", codec.int(), get: fn(v) { v.1 })
    codec.success(#(a,b))
  }
  io.println(string.inspect(codec.decode_json(duplicate, "{\"x\":1,\"extra\":2}")))
  let assert Ok(raw) = value.parse("{\"x\":1,\"extra\":2}", value.default_limits())
  io.println(string.inspect(codec.decode(duplicate, raw)))
  io.println(string.inspect(codec.check(duplicate)))
}
